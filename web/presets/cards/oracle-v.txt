# Oracle V: its factory presets, one block per preset (web/src/preset-cards.ts; npm run presets -- check).
# "preset: NAME" starts one (up to 24 characters; the FM-1 shows 16). Under it, one row per line: "row = value".
# A row not listed keeps Init's value. A row is "section / label", or the label alone when no other row has it.
# A number is in the row's unit; a choice row takes a name; a switch takes on or off. Init and the ranges:
#
# voice
#   voices = 5                         1..16
#   unison = off                       off, on, share
#   glide = 0.001 s                    0.001..5 s
# osc A
#   osc A / wave = saw                 sine, triangle, saw, square, sharktooth, reverse saw
#   pw = 0.5                           0.05..0.95
#   sync = 0                           -4..4
# osc B
#   osc B / wave = saw                 sine, triangle, saw, square, sharktooth, reverse saw
#   freq = 0 st                        -48..48 st
#   fine = 7 ct                        -100..100 ct
# mixer
#   A level = 0.55                     0..1
#   B level = 0.45                     0..1
#   noise = 0                          0..1
#   out: a screen, not a value
# filter
#   cutoff = 600 Hz                    20..16000 Hz
#   res = 0.2                          0..1
#   env amt = 3.5 oct                  0..4 oct
#   kbd track = off                    off, on
#   filter env: a screen, not a value
#   filter / a = 0.01 s                0.001..5 s
#   filter / d = 0.6 s                 0.001..5 s
#   filter / s = 0.35                  0..1
#   filter / r = 0.4 s                 0.001..8 s
#   filter: a screen, not a value
# amp
#   amp env: a screen, not a value
#   amp / a = 0.01 s                   0.001..5 s
#   amp / d = 0.5 s                    0.001..5 s
#   amp / s = 0.8                      0..1
#   amp / r = 0.4 s                    0.001..8 s
# mod
#   pmod env→A = 0                     -4..4
#   pmod B→A = 0                       0..8
#   lfo rate = 5 Hz                    0.02..40 Hz
#   lfo wave = triangle                sine, triangle, saw, square, S&H
#   vibrato = 0.24                     -4..4
#   lfo / osc A: a screen, not a value
