# Sallen 20: its factory presets, one block per preset (web/src/preset-cards.ts; npm run presets -- check).
# "preset: NAME" starts one (up to 24 characters; the FM-1 shows 16). Under it, one row per line: "row = value".
# A row not listed keeps Init's value. A row is "section / label", or the label alone when no other row has it.
# A number is in the row's unit; a choice row takes a name; a switch takes on or off. Init and the ranges:
# The level row (output / volume) gen-presets sets in every preset, by loudness (#448): leave it to it.
#
# vco
#   portamento = 0.001 s               0.001..5 s
#   master tune = 0                    -4..4
#   vco 1 scale = 8'                   32', 16', 8', 4'
#   vco 1 wave = SAW                   TRIANGLE, SAW, PULSE, NOISE
#   vco 1 pw = 0.5                     0.05..0.95
#   vco 2 scale = 8'                   16', 8', 4', 2'
#   vco 2 wave = SQUARE                SAW, SQUARE, NARROW, RING
#   vco 2 pitch = 7 st                 -48..48 st
#   vco 2 fine = 0 ct                  -100..100 ct
# freq mod
#   freq mod / mg/t.ext = 0            -4..4
#   eg1/ext = 0                        -4..4
# vco mixer
#   vco 1 level = 0.5                  0..1
#   vco 2 level = 0                    0..1
# highpass
#   highpass / cutoff = 60 Hz          20..16000 Hz
#   highpass / peak = 0.2              0..1
#   highpass / mg/t.ext = 0            -4..4
#   highpass / eg2/ext = 0             -4..4
# lowpass
#   lowpass / cutoff = 1600 Hz         20..16000 Hz
#   lowpass / peak = 0.45              0..1
#   lowpass / mg/t.ext = 0             -4..4
#   lowpass / eg2/ext = 2.5            -4..4
# eg 1
#   eg 1: a screen, not a value
#   delay = 0.001 s                    0.001..2 s
#   eg 1 / attack = 0.001 s            0.001..5 s
#   eg 1 / release = 0.3 s             0.001..8 s
# eg 2
#   eg 2: a screen, not a value
#   hold = 0.001 s                     0.001..2 s
#   eg 2 / attack = 0.005 s            0.001..5 s
#   decay = 0.4 s                      0.001..5 s
#   sustain = 0.45                     0..1
#   eg 2 / release = 0.3 s             0.001..8 s
# mg
#   frequency = 5 Hz                   0.02..40 Hz
#   waveform = 0.5                     0..1
# patch panel
