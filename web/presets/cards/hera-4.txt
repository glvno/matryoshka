# Hera 4: its factory presets, one block per preset (web/src/preset-cards.ts; npm run presets -- check).
# "preset: NAME" starts one (up to 24 characters; the FM-1 shows 16). Under it, one row per line: "row = value".
# A row not listed keeps Init's value. A row is "section / label", or the label alone when no other row has it.
# A number is in the row's unit; a choice row takes a name; a switch takes on or off. Init and the ranges:
#
# lfo
#   rate = 3 Hz                        0.02..40 Hz
#   delay time = 0.001 s               0.001..5 s
# dco
#   range = 8'                         16', 8', 4'
#   dco / lfo = 0                      -4..4
#   pwm = 0.3                          0..1
#   pwm mode = LFO                     LFO, MAN
#   pulse = OFF                        OFF, ON
#   saw = ON                           OFF, ON
#   sub = 0.25                         0..1
#   noise = 0                          0..1
# hpf
#   hpf / freq = 1                     0, 1, 2, 3
# vcf
#   vcf / freq = 1400 Hz               20..16000 Hz
#   res = 0.15                         0..1
#   polarity = POSITIVE                POSITIVE, NEGATIVE
#   env = 2.2                          -4..4
#   vcf / lfo = 0                      -4..4
#   kybd = 1                           -4..4
# vca
#   mode = ENV                         ENV, GATE
#   level = 1                          0..2
# env
#   a = 0.01 s                         0.001..5 s
#   d = 0.3333 s                       0.001..5 s
#   s = 0.6                            0..1
#   r = 0.2083 s                       0.001..8 s
# chorus
#   chorus = I                         OFF, I, II
# bender
#   dco = 0.1667                       -4..4
#   vcf = 0                            -4..4
#   bender / lfo = 0.04                -4..4
# portamento
#   portamento = 0.2 s                 0.001..5 s
#   porta switch = OFF                 OFF, ON
# keys
#   assign mode = POLY 1               POLY 1, POLY 2, UNISON
#   key transpose = 0                  -4..4
#   tune = 0                           -4..4
