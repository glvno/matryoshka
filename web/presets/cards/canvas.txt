# Canvas: its factory presets, one block per preset (web/src/preset-cards.ts; npm run presets -- check).
# "preset: NAME" starts one (up to 24 characters; the FM-1 shows 16). Under it, one row per line: "row = value".
# A row not listed keeps Init's value. A row is "section / label", or the label alone when no other row has it.
# A number is in the row's unit; a choice row takes a name; a switch takes on or off. Init and the ranges:
# The level rows (output / chan a, output / chan b) gen-presets sets in every preset, by loudness (#448): leave them to it.
#
# sequencer
#   sequencer / trigger = off          keyboard, pulser, off
#   stages = 5                         3, 4, 5
#   voltage 1 = 0                      -1..1
#   voltage 2 = 0.2                    -1..1
#   voltage 3 = 0.4                    -1..1
#   voltage 4 = 0.6                    -1..1
#   voltage 5 = 0.8                    -1..1
#   pulse 1 = 0                        -1..1
#   pulse 2 = 0                        -1..1
#   pulse 3 = 0                        -1..1
#   pulse 4 = 0                        -1..1
#   pulse 5 = 0                        -1..1
# envelope
#   envelope: a screen, not a value
#   envelope / trigger = keyboard      keyboard, pulser, sequencer
#   envelope / mode = transient        sustained, transient, off
#   attack = 0.003 s                   0.001..10 s
#   sustain = 0.02 s                   0.001..10 s
#   decay = 0.6 s                      0.001..10 s
# pulser
#   pulser / trigger = self            keyboard, self, sequencer
#   pulser / mode = triggered          triggered, off, once
#   period = 0.33 s                    0.001..10 s
#   period proc = 0                    -4..4
# random
#   random / trigger = pulser          keyboard, pulser, sequencer
# mod osc
#   mod osc / keyboard = on            on, off
#   range = + hi                       - high, + hi, low
#   frequency = 0.5667                 0..1
#   mod osc / fine tune = 0.5          0..1
#   waveshape = triangle               sine, triangle, saw, square, sharktooth, reverse saw
#   modulation = f.m. osc              bal. ext, f.m. osc, a.m. osc
#   freq proc = 0                      -4..4
#   index = 1                          -4..4
#   index proc = 0                     -4..4
# complex osc
#   complex osc / keyboard = on        on, off
#   polarity = +                       -, +
#   pitch = 0 st                       -48..48 st
#   complex osc / fine tune = 0.5      0..1
#   pitch proc = 0                     0..2
#   timbre wave = saw                  sine, triangle, saw, square, sharktooth, reverse saw
#   timbre mix = 0                     0..1
#   timbre = 2.2                       1..32
#   timbre proc = 0                    -4..4
# lopass gate
#   mode 1 = combination               lopass, combination, vca
#   level 1 = 0                        -4..4
#   level 1 proc = 0.75                -4..4
#   source 2 = gate 1                  preamp, mod osc, gate 1
#   mode 2 = combination               lopass, combination, vca
#   level 2 = 0                        -4..4
#   level 2 proc = 0.75                -4..4
# output
#   chan a = 1.81                      -4..4
#   chan b = 0                         -4..4
#   reverb = 0.15                      0..1
# keyboard
#   portamento = 0.001 s               0.001..5 s
#   porta proc = 0                     -4..4
#   add to main = none                 octaves, preset, none
#   plate = 1                          1, 2, 3, 4
#   preset 1 = 0                       -4..4
#   preset 2 = 0.25                    -4..4
#   preset 3 = 0.5                     -4..4
#   preset 4 = 1                       -4..4
# patch field
#   pitch in = sequencer               main, pressure, sequencer, envelope, pulser, random 1, mod cv, patch
#   freq in = mod cv                   main, pressure, sequencer, envelope, pulser, random 1, mod cv, patch
#   index in = envelope                main, pressure, sequencer, envelope, pulser, random 1, mod cv, patch
#   timbre in = random 1               main, pressure, sequencer, envelope, pulser, random 1, mod cv, patch
#   level 1 in = envelope              main, pressure, sequencer, envelope, pulser, random 1, mod cv, patch
#   level 2 in = envelope              main, pressure, sequencer, envelope, pulser, random 1, mod cv, patch
#   period in = main                   main, pressure, sequencer, envelope, pulser, random 1, mod cv, patch
#   porta in = pressure                main, pressure, sequencer, envelope, pulser, random 1, mod cv, patch
#   inverter in = envelope             main, pressure, sequencer, envelope, pulser, random 1, mod cv, patch
#   out: a screen, not a value
