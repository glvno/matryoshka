# Monolith D: its factory presets, one block per preset (web/src/preset-cards.ts; npm run presets -- check).
# "preset: NAME" starts one (up to 24 characters; the FM-1 shows 16). Under it, one row per line: "row = value".
# A row not listed keeps Init's value. A row is "section / label", or the label alone when no other row has it.
# A number is in the row's unit; a choice row takes a name; a switch takes on or off. Init and the ranges:
# The level row (output / volume) gen-presets sets in every preset, by loudness (#448): leave it to it.
#
# controllers
#   tune = 0                           -4..4
#   glide = 0.04 s                     0.001..5 s
#   glide on = OFF                     OFF, ON
#   mod mix = 0                        0..1
#   osc mod = OFF                      OFF, ON
#   osc 3 ctrl = ON                    OFF, ON
#   decay on = ON                      OFF, ON
# oscillators
#   osc 1 range = 8'                   LO, 32', 16', 8', 4', 2'
#   osc 1 wave = SAW                   TRIANGLE, TRI/SAW, SAW, SQUARE, WIDE RECT, NARROW RECT
#   osc 2 range = 8'                   LO, 32', 16', 8', 4', 2'
#   osc 2 freq = 0 st                  -48..48 st
#   osc 2 fine = 6 ct                  -100..100 ct
#   osc 2 wave = SAW                   TRIANGLE, TRI/SAW, SAW, SQUARE, WIDE RECT, NARROW RECT
#   osc 3 range = 16'                  LO, 32', 16', 8', 4', 2'
#   osc 3 freq = 0 st                  -48..48 st
#   osc 3 fine = -4 ct                 -100..100 ct
#   osc 3 wave = SQUARE                TRIANGLE, REVERSE SAW, SAW, SQUARE, WIDE RECT, NARROW RECT
# mixer
#   osc 1 = 0.45                       0..1
#   osc 2 = 0.4                        0..1
#   osc 3 = 0.3                        0..1
#   noise = 0                          0..1
#   white/pink = WHITE                 WHITE, PINK
# filter
#   cutoff = 700 Hz                    20..16000 Hz
#   emphasis = 0.3                     0..1
#   contour = 3                        -4..4
#   kbd control = 1+2                  OFF, 1, 2, 1+2
#   filter mod = ON                    OFF, ON
# contours
#   filter contour: a screen, not a value
#   contours / attack (1) = 0.005 s    0.001..5 s
#   contours / decay (1) = 0.72        0..1
#   contours / sustain (1) = 0.35      0..1
